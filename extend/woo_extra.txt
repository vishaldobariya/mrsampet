//Add custom <a> wrapper with query params
            add_action( 'woocommerce_before_shop_loop_item', 'custom_loop_product_link_open_with_params', 10 );
            add_action( 'woocommerce_after_shop_loop_item', 'custom_loop_product_link_close_with_params', 5 );
            function custom_loop_product_link_open_with_params() {
                global $product;
            
                // Start with base permalink
                $product_url = get_permalink( $product->get_id() );
            
                // Add query parameters if present
                $params = array_filter([
                    'plan'    => isset($_GET['plan'])    ? sanitize_text_field($_GET['plan'])    : '',
                    'visit'   => isset($_GET['visit'])   ? sanitize_text_field($_GET['visit'])   : '',
                    'pet'     => isset($_GET['pet'])     ? sanitize_text_field($_GET['pet'])     : '',
                    'vaccine' => isset($_GET['vaccine']) ? sanitize_text_field($_GET['vaccine']) : '',
                ]);
            
                if ( !empty($params) ) {
                    $product_url = add_query_arg( $params, $product_url );
                }
            
                echo '<a href="' . esc_url( $product_url ) . '" class="woocommerce-LoopProduct-link woocommerce-loop-product__link">';
            }
            
            function custom_loop_product_link_close_with_params() {
                echo '</a>';
            }

            add_action( 'woocommerce_shop_loop_item_title', 'custom_replace_title_link_start', 9 );
            function custom_replace_title_link_start() {
                ob_start();
            }
            
            add_action( 'woocommerce_shop_loop_item_title', 'custom_replace_title_link_end', 11 );
            function custom_replace_title_link_end() {
                $original_title_html = ob_get_clean();
            
                global $product;
            
                $base_url = get_permalink( $product->get_id() );
                $params = array_filter([
                    'plan'    => isset($_GET['plan'])    ? sanitize_text_field($_GET['plan'])    : '',
                    'visit'   => isset($_GET['visit'])   ? sanitize_text_field($_GET['visit'])   : '',
                    'pet'     => isset($_GET['pet'])     ? sanitize_text_field($_GET['pet'])     : '',
                    'vaccine' => isset($_GET['vaccine']) ? sanitize_text_field($_GET['vaccine']) : '',
                ]);
                $custom_url = !empty( $params ) ? add_query_arg( $params, $base_url ) : $base_url;
            
                // Replace the href attribute inside the <a> tag
                $modified_title_html = preg_replace(
                    '#href="[^"]*"#',
                    'href="' . esc_url( $custom_url ) . '"',
                    $original_title_html
                );
            
                echo $modified_title_html;
            }
